// Pi1551 options.txt — copy to the SD root next to kernel.img.
// Settings -> Load settings reloads runtime values; Save settings writes only
// changed values, preserving existing lines, comments and unrelated options.
// Hardware, ROM filenames and other startup-only fields require a reboot.

// Drive: values editable under Settings -> Drive.
deviceID = 8
RAMBOard = 1
IgnoreReset = 0 // 0 = REACT to computer RESET; 1 = IGNORE
OnResetChangeToStartingFolder = 1 // RESET returns the SD path to /1551, even with ignoreReset = 1.
skipMotorSpinUpDelay = 1
IdleSleep = 1 // Pause D64 emulation only when the ROM and host bus are idle.
// S after the track number (e.g. 18.0S) means the drive is actually sleeping.
// G64, active transfers and unsupported/custom ROM idle loops do not sleep.

// Configured DOS slots: Settings -> ROM selects a loaded slot immediately.
// There is no longpress ROM selector and no ROM switching through the Browser.
// ROM files are searched in the SD root and /roms; missing/invalid slots are omitted.
ROM1 = dos1551.bin
ROM2 = dos1551-ram.bin
ROM3 = super_dos_1551.rom
ROM4 = super_dos_ram.bin
//ROM5 = custom_dos.bin
// Legacy ROM1551 is not the source of the menu's slot list.

// Browser.
LowercaseBrowseModeFilenames = 0
DisableSD2IECCommands = 0 // 0 = SD2IEC Cmds. ON; 1 = SD2IEC Cmds. OFF
autoMountEnabled = 1
// AutoMountImage is resolved under /1551 unless given an absolute path.
// It accepts a supported disk/container/PRG or a nonempty .lst fliplist.
// Empty, missing or invalid media behaves as auto mount OFF and hides the toggle.
//AutoMountImage = mygame.d64
//AutoMountImage = lists/favorites.lst
StarFileName = boot.t2sd // DLOAD"*" target; edited in this file, not on OLED.

// Disk -> New image opens a format selector and an explicit Create image action.
// New images are NEWIMAGE1.d64/g64, NEWIMAGE2.d64/g64, etc. in the current folder.
// Creation never mounts the new image or changes the fliplist automatically.
NewDiskType = d64 // d64 or g64; the selector starts with this value.

// Tape: Settings -> Tape -> Motor ON.
// 1 = ALWAYS, useful for computers that cannot drive MOTOR directly.
// 0 = COMP., follows GPIO6: LOW = running, HIGH = stopped.
// tapeAutoplay = 1 presses PLAY automatically when a TAP is loaded; 0 loads stopped.
tapeMotorAlwaysOn = 1
tapeAutoplay = 1

// Settings -> Sound. The SOUND wire is fixed to GPIO13 (physical pin 33).
// GPIO18 is TCBM data; GPIO19 is tape READ. Neither is selectable for sound.
SoundEnabled = 1 // Sound ON/OFF, independent of the output method below.
SoundOnGPIO = 1 // Mode: 1 = BUZZER (square wave), 0 = SAMPLE (embedded PWM/DMA click).
SoundOnGPIODuration = 500 // Buzzer duration in milliseconds.
SoundOnGPIOFreq = 1200 // Buzzer frequency in Hz.
// Freq/Duration are shown only in BUZZER mode; SAMPLE has a fixed clip length.
// Mode and enable can change runtime; both methods use the same GPIO13.

// Settings -> Display. OLED layout is always 16 columns x 8 rows on 128x64.
DisplayTemperature = 1 // Temp ON/OFF controls the header; Status always shows TEMP.
i2cLcdOnContrast = 128 // Active brightness 1..255; zero is clamped to 1 at startup.
oledDimming = 1
i2cLcdDimTime = 16 // Seconds since last UI input; 0 disables automatic dimming.
i2cLcdDimContrast = 16 // Idle brightness 0..255; 0 switches the panel OFF until input.
i2cLcdFlip = 0 // 0 = normal, 1 = rotate 180 degrees.
i2cLcdUseCBMChar = 1 // Font: 1 = C64 chargen, 0 = native 8x8 ASCII (also the fallback).
scrollHighlightRate = 0.10 // Seconds per scroll step; smaller is faster.
// oledFont and oledScroll are obsolete and ignored; use the two keys above.

// Hardware/startup-only settings: not available as runtime menu choices.
LCDName = ssd1306_128x64
//LCDName = sh1106_128x64
// I2C bus 1: SDA = GPIO2 (physical pin 3), SCL = GPIO3 (physical pin 5).
//i2cLcdAddress = 60 // 0x3c; use 61 for 0x3d
//i2cRtcAddress = 104 // 0x68 (default for DS3231/DS1307); or 0 to disable RTC.
//i2cRtcType = ds3231 // or ds1307
i2cScan = 1 // Startup scan; results are printed on HDMI.
ChargenFont = chargen.bin // SD root or /roms; ASCII is used if no valid chargen loads.
LcdLogoName = 1541classic // Startup logo: 1541classic, 1541ii or a .raw file.
QuickBoot = 1 // Skip the startup pause.
ShowOptions = 1 // Print parsed options on the HDMI startup screen.

// Fixed encoder wiring: A = GPIO17/SW3, B = GPIO22/SW2, push = GPIO27/SW1.
// Rotate to navigate; push to select; Insert adds/removes fliplist entries;
// Back and ".." return to the parent. Settings -> Encoder changes Reverse.
RotaryEncoderEnable = 1
RotaryEncoderInvert = 0

// Legacy HDMI/debug settings; they do not add OLED menu/status rows.
GraphIEC = 0
DisplayPC = 0
DisplayTCBMDebug = 0
