# Pi1551 for Raspberry Pi 2B v1.1 (BCM2836 / Cortex-A7).
# On an existing Pi1551 SD card, replace kernel.img and config.txt with
# the files from this ZIP. Keep the other files and interface wiring unchanged.
#
# Start at 900 MHz.
# To improve performance, change arm_freq=900 below to arm_freq=1000.
# If stable, you can try 1100 MHz, then 1200 MHz.
# Higher clocks are not guaranteed. If it freezes or produces errors,
# return to the last stable setting. If the Pi will not boot, insert the
# SD card into a computer and restore 900 MHz. Use good power and cooling.
# Keep force_turbo=1. Change only arm_freq; leave the other settings as they are.
kernel=kernel.img
kernel_address=0x1f00000
arm_64bit=0
force_turbo=1
arm_freq=900
core_freq=250
temp_soft_limit=70
dtparam=spi=off
force_eeprom_read=0
disable_pvt=1
